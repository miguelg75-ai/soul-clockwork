    class AudioSynth {

      constructor() {

        this.ctx = null;

        this.master = null;

      }

      init() {

        if (this.ctx) return;

        const AC = window.AudioContext || window.webkitAudioContext;

        if (!AC) return;

        this.ctx = new AC();

        this.master = this.ctx.createGain();

        this.master.gain.value = 0.25;

        this.master.connect(this.ctx.destination);

      }

      playShoot() {

        if (!this.ctx) return;

        const t = this.ctx.currentTime;

        const osc = this.ctx.createOscillator();

        const g = this.ctx.createGain();

        osc.type = 'square';

        osc.frequency.setValueAtTime(380, t);

        osc.frequency.exponentialRampToValueAtTime(40, t + 0.08);

        g.gain.setValueAtTime(0.4, t);

        g.gain.linearRampToValueAtTime(0, t + 0.08);

        osc.connect(g);

        g.connect(this.master);

        osc.start(t);

        osc.stop(t + 0.08);

      }

      playHit(isCrit) {

        osc.frequency.setValueAtTime(isCrit ? 160 : 220, t);

        osc.frequency.exponentialRampToValueAtTime(50, t + (isCrit ? 0.12 : 0.06));

        g.gain.setValueAtTime(isCrit ? 0.6 : 0.3, t);

    class ObjectPool {

      constructor(factory, size) {

        this.pool = new Array(size);

        this.free = new Array(size);

        this.count = size;

        for (let i = 0; i < size; i++) {

          this.pool[i] = factory();

          this.pool[i].active = false;

          this.free[i] = i;

        }

      }

      obtain() {

        if (this.count === 0) return null;

        this.count--;

        const inst = this.pool[this.free[this.count]];

        inst.active = true;

        inst.reset();

        return inst;

      }

      release(inst) {

        const idx = this.pool.indexOf(inst);

        if (idx === -1 || !inst.active) return;

        inst.active = false;

        this.free[this.count] = idx;

        this.count++;

      }

      getActive() { return this.pool; }

    }


    class SpatialGrid {

      constructor(w, h, cell = 64) {

        this.cell = cell;

        this.cols = Math.ceil(w / cell);

        this.rows = Math.ceil(h / cell);

        const total = this.cols * this.rows;

        this.grid = new Array(total).fill(0).map(() => new Array(64).fill(-1));

        this.counts = new Array(total).fill(0);

      }

      clear() { this.counts.fill(0); }

      insert(x, y, poolIdx) {

        const c = Math.floor(x / this.cell);

        const r = Math.floor(y / this.cell);

        if (c < 0 || c >= this.cols || r < 0 || r >= this.rows) return;

        const idx = r * this.cols + c;

        if (this.counts[idx] < 64) {

          this.grid[idx][this.counts[idx]] = poolIdx;

          this.counts[idx]++;

        }

      }

      query(x, y, rad, out) {

        let n = 0;

        const minC = Math.max(0, Math.floor((x - rad) / this.cell));

        const maxC = Math.min(this.cols - 1, Math.floor((x + rad) / this.cell));

        const minR = Math.max(0, Math.floor((y - rad) / this.cell));

        const maxR = Math.min(this.rows - 1, Math.floor((y + rad) / this.cell));

        for (let r = minR; r <= maxR; r++) {

          for (let c = minC; c <= maxC; c++) {

            const idx = r * this.cols + c;

            const cnt = this.counts[idx];

            for (let i = 0; i < cnt; i++) out[n++] = this.grid[idx][i];

          }

        }

        return n;

      }

    }


    class Projectile {

      constructor() { this.active = false; this.x = 0; this.y = 0; this.vx = 0; this.vy = 0; this.damage = 15; this.life = 0; }

      reset() { this.life = 0; }

      init(x, y, dx, dy, dmg) {

        this.x = x; this.y = y;

        this.vx = dx * 480; this.vy = dy * 480;

        this.damage = dmg;

      }

      update(dt, pool) {

        if (!this.active) return;

        this.x += this.vx * dt;

        this.y += this.vy * dt;

        this.life += dt;

        if (this.life > 1.8) pool.release(this);

      }

      render(ctx) {

        if (!this.active) return;

        ctx.fillStyle = '#ffcc00';

        ctx.beginPath();

        ctx.arc(this.x, this.y, 4, 0, Math.PI * 2);

        ctx.fill();

      }

    }


    class Enemy {

      constructor() { this.active = false; this.x = 0; this.y = 0; this.hp = 20; this.speed = 85; this.radius = 10; this.damage = 10; this.kx = 0; this.ky = 0; this.color = '#a855f7'; }

      reset() { this.kx = this.ky = 0; }

      init(type, x, y, diff) {

        this.x = x; this.y = y;

        if (type === 0) { this.hp = 16 * diff; this.speed = 105; this.radius = 8; this.damage = 6; this.color = '#a855f7'; }

        else if (type === 1) { this.hp = 75 * diff; this.speed = 50; this.radius = 15; this.damage = 15; this.color = '#b45309'; }

        else if (type === 2) { this.hp = 35 * diff; this.speed = 70; this.radius = 11; this.damage = 10; this.color = '#06b6d4'; }

        else { this.hp = 25 * diff; this.speed = 130; this.radius = 9; this.damage = 8; this.color = '#e11d48'; }

      }

      applyDamage(amt, px, py) {

        this.hp -= amt;

        this.kx += px * 120;

        this.ky += py * 120;

        return this.hp <= 0;

      }

      update(dt, tx, ty) {

        if (!this.active) return;

        const dx = tx - this.x;

        const dy = ty - this.y;

        const dist = Math.hypot(dx, dy) || 1;

        this.kx *= Math.max(0, 1 - dt * 6);

        this.ky *= Math.max(0, 1 - dt * 6);

        this.x += ((dx / dist) * this.speed + this.kx) * dt;

        this.y += ((dy / dist) * this.speed + this.ky) * dt;

      }

      render(ctx) {

        if (!this.active) return;

        ctx.fillStyle = this.color;

        ctx.beginPath();

        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);

        ctx.fill();

      }

    }


    class Gem {

      constructor() { this.active = false; this.x = 0; this.y = 0; this.val = 10; this.attracted = false; this.spd = 0; }

      reset() { this.attracted = false; this.spd = 0; }

      init(x, y, val) { this.x = x; this.y = y; this.val = val; }

      update(dt, px, py, rad, pool, onCollect) {

        if (!this.active) return;

        const dx = px - this.x;

        const dy = py - this.y;

        const d = Math.hypot(dx, dy);

        if (!this.attracted && d < rad) this.attracted = true;

        if (this.attracted) {

          if (d < 18) { onCollect(this.val); pool.release(this); return; }

          this.spd = Math.min(650, this.spd + 850 * dt);

          this.x += (dx / d) * this.spd * dt;

          this.y += (dy / d) * this.spd * dt;

        }

      }

      render(ctx) {

        if (!this.active) return;

        ctx.fillStyle = '#38bdf8';

        ctx.beginPath();

        ctx.arc(this.x, this.y, 3.5, 0, Math.PI * 2);

      }

    }

            const sx = this.px + Math.cos(ang) * 450;

            const sy = this.py + Math.sin(ang) * 450;

            let type = 0;

            if (min > 1.5 && Math.random() < 0.25) type = 3;

            else if (min > 3.0 && Math.random() < 0.4) type = 2;

            else if (min > 5.0 && Math.random() < 0.5) type = 1;

            e.init(type, sx, sy, 1 + min * 0.2);

          }

        }


        this.grid.clear();

        const enemies = this.enemyPool.getActive();

        for (let i = 0; i < enemies.length; i++) {

          const e = enemies[i];

          if (e.active) {

            e.update(dt, this.px, this.py);

            this.grid.insert(e.x, e.y, i);

            const d = Math.hypot(this.px - e.x, this.py - e.y);

            if (d < e.radius + 12) {

              this.hp -= e.damage * dt;

              if (this.hp <= 0) { this.hp = 0; this.triggerGameOver(true); return; }

            }

          }

        }


        this.updateWeapons(dt);


        const projs = this.projPool.getActive();

        for (let i = 0; i < projs.length; i++) {

          const p = projs[i];

          if (!p.active) continue;

          p.update(dt, this.projPool);

          if (!p.active) continue;


          const count = this.grid.query(p.x, p.y, 24, this.nearbyBuf);

          for (let k = 0; k < count; k++) {

            const e = enemies[this.nearbyBuf[k]];

            if (e && e.active && Math.hypot(e.x - p.x, e.y - p.y) < e.radius + 4) {

              const isCrit = Math.random() < 0.15;

              const dmg = isCrit ? p.damage * 1.8 : p.damage;

              const dead = e.applyDamage(dmg, p.vx / 480, p.vy / 480);

              this.audio.playHit(isCrit);

              this.textPool.obtain()?.init(e.x, e.y, dmg, isCrit);

              if (isCrit) this.trauma = Math.min(1.0, this.trauma + 0.15);

              if (dead) this.killEnemy(e);

              this.projPool.release(p);

              break;

            }

          }

        }


        const gems = this.gemPool.getActive();

        const pickupRad = 70 * (1 + (this.passives.cobalt_magnet || 0) * 0.25);

        for (let i = 0; i < gems.length; i++) {

          gems[i].update(dt, this.px, this.py, pickupRad, this.gemPool, (val) => {

            this.audio.playGem();

            this.addXP(val);

          });

        }

        const texts = this.textPool.getActive();

        for (let i = 0; i < texts.length; i++) texts[i].update(dt, this.textPool);


        if (this.trauma > 0) this.trauma = Math.max(0, this.trauma - dt * 1.8);


        this.updateHUD();

      }


      updateWeapons(dt) {

        this.shootTimer += dt;

        const isGatling = !!this.weapons.soul_gatling;

        const shootRate = isGatling ? 0.12 : Math.max(0.2, 0.45 - (this.weapons.brass_revolver || 1) * 0.03);

        if (this.shootTimer >= shootRate) {

          this.shootTimer = 0;

          const p = this.projPool.obtain();

          if (p) {

            p.init(this.px, this.py, this.facing.x, this.facing.y, isGatling ? 40 : 18 + (this.weapons.brass_revolver || 1) * 4);

            this.audio.playShoot();

          }

        }


        const vortexLvl = this.weapons.steam_vortex || 0;

        const isFurnace = !!this.weapons.aether_furnace;

        if (vortexLvl > 0 || isFurnace) {

          this.vortexTimer += dt;

          if (this.vortexTimer >= 0.25) {

            this.vortexTimer = 0;

            const rad = isFurnace ? 130 : 60 + vortexLvl * 8;

            const dmg = (isFurnace ? 35 : 6 + vortexLvl * 3);

            const count = this.grid.query(this.px, this.py, rad, this.nearbyBuf);

            const enemies = this.enemyPool.getActive();

            for (let i = 0; i < count; i++) {

              const e = enemies[this.nearbyBuf[i]];

              if (e && e.active && Math.hypot(e.x - this.px, e.y - this.py) < rad + e.radius) {

                if (e.applyDamage(dmg, 0, 0)) this.killEnemy(e);

              }

            }

          }

        }


        this.pendulumAngle = (this.pendulumAngle + dt * 4) % (Math.PI * 2);


        const teslaLvl = this.weapons.galvanic_strike || 0;

        const isTesla = !!this.weapons.tesla_judgement;

        if (teslaLvl > 0 || isTesla) {

          this.teslaTimer += dt;

          if (this.teslaTimer >= (isTesla ? 0.9 : 2.2)) {

            this.teslaTimer = 0;

            const enemies = this.enemyPool.getActive();

            let hits = 0;

            for (let i = 0; i < enemies.length; i++) {

              const e = enemies[i];

              if (e && e.active && Math.hypot(e.x - this.px, e.y - this.py) < 400) {

                if (e.applyDamage(isTesla ? 70 : 35, 0, 0)) this.killEnemy(e);

                this.textPool.obtain()?.init(e.x, e.y, isTesla ? 70 : 35, true);

                hits++;

                if (hits >= (isTesla ? 6 : 2)) break;

              }

            }

          }

        }

      }


      killEnemy(e) {

        this.kills++;

        this.gold += 2;

        const gem = this.gemPool.obtain();

        if (gem) gem.init(e.x, e.y, 15);

        this.enemyPool.release(e);

      }


      addXP(amt) {

        this.xp += amt;

        if (this.xp >= this.xpNext) {

          this.xp -= this.xpNext;

          this.level++;

          this.xpNext = Math.floor(this.xpNext * 1.2);

          this.audio.playLevelUp();

          this.showLevelUp();

        }

      }


      showLevelUp() {

        this.isPaused = true;

        const container = document.getElementById('choices-container');

        container.innerHTML = '';

        const choices = [

          { id: 'brass_revolver', name: 'Revólver de Latón', desc: '+Daño y velocidad de disparo.' },

          { id: 'steam_vortex', name: 'Vórtice de Vapor', desc: 'Aura circular de calor continuo.' },

          { id: 'galvanic_strike', name: 'Descarga Galvánica', desc: 'Rayos eléctricos a distancia.' },

          { id: 'volatile_oil', name: 'Aceite Volátil', desc: '+15% Daño global (Pasiva).' }

        ];


        choices.slice(0, 3).forEach(c => {

          const card = document.createElement('div');

          card.className = 'upgrade-card';

          card.innerHTML = `<div style="font-weight:bold; color:#fcd34d">${c.name}</div><div style="font-size:12px; color:#cbd5e1">${c.desc}</div>`;

          card.onclick = () => {

            if (c.id === 'volatile_oil') this.passives[c.id] = (this.passives[c.id] || 0) + 1;

            else this.weapons[c.id] = (this.weapons[c.id] || 0) + 1;

            document.getElementById('levelup-ui').style.display = 'none';

            this.isPaused = false;

          };

          container.appendChild(card);

        });

        document.getElementById('levelup-ui').style.display = 'flex';

      }


      triggerGameOver(died) {

        this.isPaused = true;

        const ui = document.getElementById('gameover-ui');

        document.getElementById('go-title').innerText = died ? 'NÚCLEO DESTRUIDO' : '¡VICTORIA!';

        document.getElementById('go-stats').innerHTML = `Tiempo sobrevivido: <b>${this.formatTime(this.gameTime)}</b><br>Espectros purgados: <b>${this.kills}</b><br>Oro obtenido: <b>🪙 ${this.gold}</b>`;

        

        const btnRevive = document.getElementById('btn-revive');

        btnRevive.style.display = (died && !this.hasRevived) ? 'block' : 'none';

        btnRevive.onclick = () => {

          this.hasRevived = true;

          this.hp = this.maxHp * 0.5;

          ui.style.display = 'none';

          this.isPaused = false;

        };


        document.getElementById('btn-double').onclick = () => {

          this.gold *= 2;

          this.resetRun();

        };

        document.getElementById('btn-claim').onclick = () => this.resetRun();


        ui.style.display = 'flex';

      }


      resetRun() {

        location.reload();

      }


      updateHUD() {

        document.getElementById('xp-fill').style.width = `${(this.xp / this.xpNext) * 100}%`;

        document.getElementById('lvl-txt').innerText = `NIVEL ${this.level}`;

        document.getElementById('time-txt').innerText = this.formatTime(this.gameTime);

        document.getElementById('hp-txt').innerText = `HP ${Math.ceil(this.hp)}/${this.maxHp}`;

        document.getElementById('gold-txt').innerText = `🪙 ${this.gold}`;

      }


      formatTime(s) {

        const m = Math.floor(s / 60).toString().padStart(2, '0');

        const sec = Math.floor(s % 60).toString().padStart(2, '0');

        return `${m}:${sec}`;

      }


      render() {

        const ctx = this.ctx;

        ctx.save();


        if (this.trauma > 0) {

          const shk = this.trauma * this.trauma * 12;

          ctx.translate((Math.random() * 2 - 1) * shk, (Math.random() * 2 - 1) * shk);

        }


        ctx.fillStyle = '#0b0f19';

        ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);


        ctx.save();

        ctx.translate(this.canvas.width / 2 - this.px, this.canvas.height / 2 - this.py);


        ctx.strokeStyle = 'rgba(30, 41, 59, 0.4)';

        ctx.lineWidth = 1;

        const startX = Math.floor((this.px - this.canvas.width / 2) / 64) * 64;

        const startY = Math.floor((this.py - this.canvas.height / 2) / 64) * 64;

        for (let x = startX; x < startX + this.canvas.width + 128; x += 64) {

          ctx.beginPath(); ctx.moveTo(x, startY); ctx.lineTo(x, startY + this.canvas.height + 128); ctx.stroke();

        }

        for (let y = startY; y < startY + this.canvas.height + 128; y += 64) {

          ctx.beginPath(); ctx.moveTo(startX, y); ctx.lineTo(startX + this.canvas.width + 128, y); ctx.stroke();

        }


        const vortexLvl = this.weapons.steam_vortex || 0;

        if (vortexLvl > 0 || this.weapons.aether_furnace) {

          const rad = this.weapons.aether_furnace ? 130 : 60 + vortexLvl * 8;

          ctx.beginPath();

          ctx.arc(this.px, this.py, rad, 0, Math.PI * 2);

          ctx.fillStyle = 'rgba(14, 165, 233, 0.12)';

          ctx.fill();

          ctx.strokeStyle = 'rgba(56, 189, 248, 0.4)';

          ctx.stroke();

        }


        const gems = this.gemPool.getActive();

        for (let i = 0; i < gems.length; i++) gems[i].render(ctx);


        const projs = this.projPool.getActive();

        for (let i = 0; i < projs.length; i++) projs[i].render(ctx);


        const enemies = this.enemyPool.getActive();

        for (let i = 0; i < enemies.length; i++) enemies[i].render(ctx);


        ctx.fillStyle = '#4ade80';

        ctx.beginPath();

        ctx.arc(this.px, this.py, 13, 0, Math.PI * 2);

        ctx.fill();

        ctx.strokeStyle = '#ffffff';

        ctx.lineWidth = 2.5;

        ctx.beginPath();

        ctx.moveTo(this.px, this.py);

        ctx.lineTo(this.px + this.facing.x * 18, this.py + this.facing.y * 18);

        ctx.stroke();


        const texts = this.textPool.getActive();

        for (let i = 0; i < texts.length; i++) texts[i].render(ctx);


        ctx.restore();


        if (this.touchActive) {

          ctx.beginPath();

          ctx.arc(this.touchOrigin.x, this.touchOrigin.y, 50, 0, Math.PI * 2);

          ctx.fillStyle = 'rgba(15, 23, 42, 0.35)';

          ctx.fill();

          ctx.strokeStyle = 'rgba(245, 158, 11, 0.5)';

          ctx.lineWidth = 2;

          ctx.stroke();


          ctx.beginPath();

          ctx.arc(this.touchCurrent.x, this.touchCurrent.y, 20, 0, Math.PI * 2);

          ctx.fillStyle = 'rgba(245, 158, 11, 0.7)';

          ctx.fill();

        }


        ctx.restore();

      }

    }
